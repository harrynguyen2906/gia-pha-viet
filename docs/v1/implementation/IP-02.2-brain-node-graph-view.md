```typescript
// src/lib/brain-graph-transform.ts
import { BrainGraphData, BrainNode, BrainLink } from "@/types/brain-graph";

export function extractShortName(fullName: string): string {
  const parts = fullName.split(" ");
  return parts.length > 2 ? parts.slice(-2).join(" ") : fullName;
}

export function getNodeRadius(node: BrainNode, isRoot: boolean): number {
  if (isRoot) return 28;
  const base = 12;
  const bonus = Math.min(node.connectionCount * 1.5, 12);
  return node.connectionCount === 0 ? 8 : base + bonus;
}

export function getNodeColor(
  node: BrainNode,
  selectedId?: string | null,
): { fill: string; border: string } {
  if (node.id === selectedId) return NODE_COLORS.SELECTED;
  if (node.isDeceased) return NODE_COLORS.DECEASED;
  if (node.lineageRole && NODE_COLORS[node.lineageRole])
    return NODE_COLORS[node.lineageRole];
  return node.gender === "FEMALE" ? NODE_COLORS.FEMALE : NODE_COLORS.MALE;
}

export function buildBrainGraphData(
  members: Member[],
  relationships: Relationship[],
  filters?: BrainFilters,
): BrainGraphData {
  // 1. Filter members theo filters
  // 2. Map members → BrainNode[]
  // 3. Map relationships → BrainLink[] (chỉ giữ links có cả 2 nodes)
  // 4. Tính connectionCount cho mỗi node
  // 5. Return { nodes, links }
}
```

```typescript
// src/hooks/useBrainGraphData.ts
export function useBrainGraphData() {
  const { data: members } = useMembers({ limit: 1000 });
  const { data: relationships } = useRelationships();
  const filters = useLineageStore((s) => s.brainFilters);

  return useMemo(() => {
    if (!members?.data || !relationships?.data) return null;
    return buildBrainGraphData(members.data, relationships.data, filters);
  }, [members?.data, relationships?.data, filters]);
}
```

### Custom Node Painting (Canvas)

```typescript
// Canvas render cho mỗi node
const paintNode = useCallback(
  (node: BrainNode, ctx: CanvasRenderingContext2D, globalScale: number) => {
    const radius = getNodeRadius(node, isRootNode(node));
    const color = getNodeColor(node, selectedNodeId);
    const isSelected = node.id === selectedNodeId;
    const isHighlighted = highlightedNodes.has(node.id);

    // 1. Draw node circle
    ctx.beginPath();
    ctx.arc(node.x!, node.y!, radius, 0, 2 * Math.PI);
    ctx.fillStyle = color.fill;
    ctx.fill();

    // 2. Draw border
    ctx.strokeStyle = isSelected ? "#e5c17c" : color.border;
    ctx.lineWidth = isSelected ? 3 : 1.5;
    ctx.stroke();

    // 3. Glow effect khi selected
    if (isSelected || isHighlighted) {
      ctx.shadowColor = color.fill;
      ctx.shadowBlur = 15;
      ctx.beginPath();
      ctx.arc(node.x!, node.y!, radius, 0, 2 * Math.PI);
      ctx.fill();
      ctx.shadowBlur = 0;
    }

    // 4. Draw label (chỉ khi zoom >= 0.5)
    if (showLabels && globalScale >= 0.5) {
      ctx.font = `${isHighlighted ? "bold " : ""}${10 / globalScale}px "Noto Sans", sans-serif`;
      ctx.textAlign = "center";
      ctx.fillStyle = "rgba(244,235,208,0.8)";
      ctx.fillText(
        node.shortName,
        node.x!,
        node.y! + radius + 12 / globalScale,
      );
    }
  },
  [selectedNodeId, highlightedNodes, showLabels],
);
```

### Custom Link Painting

```typescript
const paintLink = useCallback(
  (link: BrainLink, ctx: CanvasRenderingContext2D) => {
    const isHighlighted = highlightedLinks.has(link);
    const isSpouse = link.type === "SPOUSE";

    ctx.beginPath();
    ctx.moveTo(link.source.x, link.source.y);
    ctx.lineTo(link.target.x, link.target.y);

    ctx.strokeStyle = isHighlighted
      ? "rgba(229,193,124,0.6)"
      : isSpouse
        ? "rgba(138,0,0,0.3)"
        : "rgba(197,160,89,0.25)";
    ctx.lineWidth = isHighlighted ? 2 : 1;

    if (isSpouse) ctx.setLineDash([4, 4]);
    ctx.stroke();
    if (isSpouse) ctx.setLineDash([]);
  },
  [highlightedLinks],
);
```

### Interaction Handlers

```typescript
// Click node → select + highlight connections
const handleNodeClick = useCallback(
  (node: BrainNode) => {
    onSelectNode(node.id);
    // Compute highlighted nodes & links
    const connectedNodes = new Set<string>();
    const connectedLinks = new Set<BrainLink>();
    graphData.links.forEach((link) => {
      const sid =
        typeof link.source === "object" ? link.source.id : link.source;
      const tid =
        typeof link.target === "object" ? link.target.id : link.target;
      if (sid === node.id || tid === node.id) {
        connectedNodes.add(sid);
        connectedNodes.add(tid);
        connectedLinks.add(link);
      }
    });
    setHighlightedNodes(connectedNodes);
    setHighlightedLinks(connectedLinks);
  },
  [graphData, onSelectNode],
);

// Double-click → zoom to node
const handleNodeDoubleClick = useCallback((node: BrainNode) => {
  graphRef.current?.centerAt(node.x, node.y, 500);
  graphRef.current?.zoom(2, 500);
}, []);

// Click background → deselect
const handleBackgroundClick = useCallback(() => {
  onSelectNode(null);
  setHighlightedNodes(new Set());
  setHighlightedLinks(new Set());
}, [onSelectNode]);
```


**Rendering logic:**

**Drag viewport:** Optional v1 — mousedown on viewport rect → mousemove updates main graph center.
